    1: /**
    2:  * CallContext.tsx
    3:  *
    4:  * Central call state machine. Single source of truth for all call state.
    5:  *
    6:  * States: IDLE â†’ INITIATING â†’ RINGING â†’ CONNECTING â†’ ACTIVE â†’ ENDING â†’ IDLE
    7:  *                                                      â†•         â†•
    8:  *                                               DEGRADED    RECONNECTING
    9:  *         Any state â†’ FAILED â†’ IDLE
   10:  *
   11:  * DEGRADED: poor network, media still flowing but quality reduced.
   12:  * RECONNECTING: network dropped, attempting recovery.
   13:  *
   14:  * Rules:
   15:  *  - All transitions go through transitionTo()
   16:  *  - No direct setState from UI
   17:  *  - No boolean flags controlling logic â€” only state drives behavior
   18:  *  - Audio managed exclusively via audioService
   19:  *  - One subscription per call, always cleaned on ENDING/FAILED
   20:  *  - 45s timeout in RINGING state
   21:  *  - RECONNECTING on network drop, FAILED after 30s timeout
   22:  *  - Action lock prevents rapid duplicate user actions
   23:  *  - Timer uses wall-clock timestamps, not interval increments
   24:  *
   25:  * Supports 1-on-1 and group calls.
   26:  * Does NOT change callService.ts or the DB model.
   27:  */

  121: async function playSound(type: SoundType): Promise<boolean> {
  122:   // Guard: already playing this sound
  123:   if (currentType === type && currentSound !== null) {
  124:     console.log('[AudioService] already playing', type);
  125:     return true;
  126:   }
  127: 
  128:   // Guard: stop in progress
  129:   if (stopping) {
  130:     console.log('[AudioService] stop in progress, skipping play', type);
  131:     return false;
  132:   }
  133: 
  134:   // Stop any existing sound first
  135:   await doStop(false);
  136: 
  137:   const asset = type === 'ringtone' ? ringtoneAsset : ringbackAsset;
  138:   if (!asset) {
  139:     console.log('[AudioService] asset not found for', type);
  140:     return false;
  141:   }
  142: 
  143:   // Configure audio session
  144:   const configured = await configureAudioSession();
  145:   if (!configured) {
  146:     console.log('[AudioService] audio session config failed for', type);
  147:     return false;
  148:   }
  149: 
  150:   try {
  151:     const sound = createAudioPlayer(asset);
  152:     sound.loop = true;
  153:     sound.volume = type === 'ringtone' ? 1.0 : 0.5;
  154:     sound.play();

  157:     currentType = type;
  158:     console.log('[AudioService] playing', type);

  160:     // Start recovery loop
  161:     startRecovery(type);
  162: 
  163:     return true;
  164:   } catch (e) {
  165:     console.log('[AudioService] play error for', type, e);
  166:     currentSound = null;
  167:     currentType = null;
  168:     return false;
  169:   }

  172: /**
  173:  * Recovery loop: checks every 1.5s if sound is still playing.
  174:  * If interrupted (by system, other app, or audio session change),
  175:  * reconfigures audio session and restarts playback.
  176:  */
  177: function startRecovery(type: SoundType): void {
  178:   stopRecovery();
  179: 
  180:   recoveryInterval = setInterval(async () => {
  181:     if (!currentSound || currentType !== type || stopping) {
  182:       stopRecovery();
  183:       return;
  184:     }
  185: 
  186:     try {
  187:       const status = { isLoaded: true, isPlaying: currentSound.playing };
  188:       if (!status.isLoaded) {
  189:         console.log('[AudioService] recovery: sound unloaded, recreating', type);
  190:         await doStop(false);
  191:         await playSound(type);
  192:         return;
  193:       }
  194:       if (!status.isPlaying) {
  195:         console.log('[AudioService] recovery: sound paused, restarting', type);
  196:         await configureAudioSession();
  197:         currentSound.play();
  198:       }
  199:     } catch (e) {
  200:       console.log('[AudioService] recovery: error, recreating', type, e);
  201:       await doStop(false);
  202:       await playSound(type);
  203:     }
  204:   }, 1500);
  205: }

  209:  */

  211:   if (recoveryInterval !== null) {
  212:     clearInterval(recoveryInterval);
  213:     recoveryInterval = null;
  214:   }
  215: }

  239:   }

  242:     await resetAudioSession();
  243:   }

  161:       clearCallNavGuard();
  162:     };
  163:   }, [userId, nav]);
  164: 
  165:   return null;

    1: // daily-get-token: create-or-get the Daily room for a call session and mint
    2: // a meeting token. Room name IS the call session id, so every participant
    3: // of a session lands in the same room â€” 1:1 and group alike.
    4: import { createClient } from 'jsr:@supabase/supabase-js@2';
    5: 
    6: const cors = {
    7:   'Access-Control-Allow-Origin': '*',
    8:   'Access-Control-Allow-Headers': 'authorization, x-client-info, apikey, content-type',
    9:   'Access-Control-Allow-Methods': 'POST, OPTIONS',
   10: };
   11: const json = (s: number, b: unknown) =>
   12:   new Response(JSON.stringify(b), { status: s, headers: { ...cors, 'Content-Type': 'application/json' } });
   13: 
   14: Deno.serve(async (req) => {
   15:   if (req.method === 'OPTIONS') return new Response('ok', { headers: cors });
   16:   try {
   17:     const DAILY_KEY = Deno.env.get('DAILY_API_KEY');
   18:     if (!DAILY_KEY) return json(500, { error: 'DAILY_API_KEY not configured' });
   19: 
   20:     const authHeader = req.headers.get('Authorization') ?? '';
   21:     const anon = createClient(Deno.env.get('SUPABASE_URL')!, Deno.env.get('SUPABASE_ANON_KEY')!, {
   22:       global: { headers: { Authorization: authHeader } },
   23:     });
   24:     const { data: userData, error: authErr } = await anon.auth.getUser();
   25:     if (authErr || !userData?.user) return json(401, { error: 'unauthorized' });
   26:     const uid = userData.user.id;
   27: 
   28:     const body = await req.json();
   29:     const sessionId: string | null = body.callSessionId || null;
   30:     const roomName: string = String(sessionId || body.roomName || '').replace(/[^a-zA-Z0-9-_]/g, '');
   31:     if (!roomName) return json(400, { error: 'no room' });
   32:     const isOwner: boolean = body.isOwner === true;
   33: 
   34:     // The caller must belong to the session: initiator, receiver, or a
   35:     // participant row (groups). Rooms are private; this is the gate.
   36:     if (sessionId) {
   37:       const svc = createClient(Deno.env.get('SUPABASE_URL')!, Deno.env.get('SUPABASE_SERVICE_ROLE_KEY')!);
   38:       const { data: sess } = await svc.from('call_sessions')
   39:         .select('id, initiator_id, receiver_id').eq('id', sessionId).maybeSingle();
   40:       if (!sess) return json(404, { error: 'call not found' });
   41:       let allowed = sess.initiator_id === uid || sess.receiver_id === uid;
   42:       if (!allowed) {
   43:         const { data: part } = await svc.from('call_participants')
   44:           .select('user_id').eq('call_session_id', sessionId).eq('user_id', uid).maybeSingle();
   45:         allowed = !!part;
   46:       }
   47:       if (!allowed) return json(403, { error: 'not a participant' });
   48:     }
   49: 
   50:     let displayName = '';
   51:     try {
   52:       const svc2 = createClient(Deno.env.get('SUPABASE_URL')!, Deno.env.get('SUPABASE_SERVICE_ROLE_KEY')!);
   53:       const { data: prof } = await svc2.from('profiles').select('full_name').eq('id', uid).maybeSingle();
   54:       displayName = String(prof?.full_name || '');
   55:     } catch { /* name is a nicety, never a blocker */ }
   56:

   59:     // create-or-get the room

   65:       const createRes = await fetch('https://api.daily.co/v1/rooms', {
   66:         method: 'POST', headers: dailyHeaders,
   67:         body: JSON.stringify({
   68:           name: roomName,
